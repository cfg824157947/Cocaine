  import pandas as pd
  import logomaker
  import matplotlib.pyplot as plt
  import os

  def plot_multiple_motifs(motif_file, motif_names, out_dir=None, figsize=(6, 2.5)):
      """
      Plot multiple motifs from a HOMER .motif file.

      Parameters
      ----------
      motif_file : str
          Path to HOMER .motif file.
      motif_names : list of str
          Motif names to extract (exact or partial match).
      out_dir : str or None
          If given, saves each motif logo as a PNG in this directory.
      figsize : tuple
          Figure size for each motif logo.
      """

      # read file once
      with open(motif_file) as f:
          lines = f.readlines()

      for motif_name in motif_names:
          pwm_lines = []
          capture = False

          for line in lines:
              line = line.strip()
              if line.startswith(">"):
                  if capture:
                      break  # stop when we reach next motif
                  if motif_name in line:  # start capturing this motif
                      capture = True
                      continue
              elif capture:
                  pwm_lines.append(line)

          if not pwm_lines:
              print(f"⚠️ Motif '{motif_name}' not found.")
              continue

          # convert PWM to DataFrame
          pwm = pd.DataFrame([list(map(float, x.split())) for x in pwm_lines],
                             columns=["A", "C", "G", "T"])
          pwm = pwm.div(pwm.sum(axis=1), axis=0)  # normalize per position

          # plot
          plt.figure(figsize=figsize)
          logomaker.Logo(pwm, color_scheme="classic")
          plt.title(motif_name, fontsize=12)
          plt.xlabel("Position")
          plt.ylabel("Probability")
          plt.tight_layout()

          # optionally save
          if out_dir:
              os.makedirs(out_dir, exist_ok=True)
              out_path = os.path.join(out_dir, f"{motif_name.replace('/', '_')}.png")
              plt.savefig(out_path, dpi=300, bbox_inches="tight")
              print(f"✅ Saved: {out_path}")

          plt.show()

          return fig, axes

  import math
  import pandas as pd
  import logomaker
  import matplotlib.pyplot as plt
  import os

  def plot_multiple_motifs(
      motif_file,
      motif_names,
      ncols=4,
      figsize_per_motif=(8, 3.5)
  ):
      """
      Plot multiple motifs from a HOMER .motif file in an (n, m) subplot grid.

      Parameters
      ----------
      motif_file : str
          Path to HOMER .motif file.
      motif_names : list of str
          Motif names to extract (exact or partial match).
      ncols : int
          Number of columns in subplot grid.
      out_dir : str or None
          If given, saves the combined figure as a PNG.
      figsize_per_motif : tuple
          Size of each subplot (width, height).
      """

      # read file once
      with open(motif_file) as f:
          lines = f.readlines()

      motifs = []

      # extract PWMs
      for motif_name in motif_names:
          pwm_lines = []
          capture = False

          for line in lines:
              line = line.strip()
              if line.startswith(">"):
                  if capture:
                      break
                  if motif_name in line:
                      capture = True
                      continue
              elif capture:
                  pwm_lines.append(line)

          if not pwm_lines:
              print(f"⚠️ Motif '{motif_name}' not found.")
              continue

          pwm = pd.DataFrame(
              [list(map(float, x.split())) for x in pwm_lines],
              columns=["A", "C", "G", "T"],
          )
          pwm = pwm.div(pwm.sum(axis=1), axis=0)
          motifs.append((motif_name, pwm))

      if not motifs:
          raise ValueError("No motifs found.")

      # grid size
      n_motifs = len(motifs)
      nrows = math.ceil(n_motifs / ncols)

      fig, axes = plt.subplots(
          nrows,
          ncols,
          figsize=(figsize_per_motif[0] * ncols,
                   figsize_per_motif[1] * nrows),
          squeeze=False
      )

      # plot motifs
      for ax, (motif_name, pwm) in zip(axes.flat, motifs):
          logomaker.Logo(pwm, ax=ax, color_scheme="classic")
          ax.set_title(motif_name)
          ax.set_xlabel("Position")
          ax.set_ylabel("Prob")

      # turn off unused axes
      for ax in axes.flat[len(motifs):]:
          ax.axis("off")

      plt.tight_layout()

      return fig, axes
