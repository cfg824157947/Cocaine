def calculate_logFC(exp_df, meta_df):
    """
    Compute log fold change (difference) between group=1 and group=0
    for a given celltype.

    Parameters
    ----------
    exp_df : pd.DataFrame
        Expression matrix (rows=genes, cols=cells).
    meta_df : pd.DataFrame
        Cell-level metadata with columns ['celltype', 'group'] at least.
    celltype : str
        The celltype of interest.

    Returns
    -------
    FC : pd.DataFrame
        A single-column DataFrame with the difference in mean expression
        for each gene in the given celltype.
    """
    aud_barcode = meta_df[meta_df['group'] == 1].index
    ctl_barcode = meta_df[meta_df['group'] == 0].index

    mean1 = exp_df[aud_barcode].mean(axis=1)# + 1e-6
    mean0 = exp_df[ctl_barcode].mean(axis=1)# + 1e-6
    FC = pd.DataFrame(mean1 - mean0, columns=['logFC'])
    return FC

def make_logFC_df(exp_df, meta_df, key_name):
    """
    Create a DataFrame of log fold changes (differences) for all genes
    in each celltype.

    Parameters
    ----------
    exp_df : pd.DataFrame
        Expression matrix (rows=genes, cols=cells).
    meta_df : pd.DataFrame
        Cell-level metadata with columns ['celltype', 'group'] at least.
    celltype_list : list
        List of celltypes to compute logFC.

    Returns
    -------
    FC_df : pd.DataFrame
        Columns are celltypes, rows are genes. Each entry is the difference
        in mean expression for group=1 vs group=0.
    """
    key_list = meta_df[key_name].unique()
    FC_df = pd.DataFrame()
    FC_dict = {}
    for key in key_list:
        idx = meta_df[meta_df[key_name] == key].index
        FC = calculate_logFC(exp_df[idx], meta_df.loc[idx])
        #FC_df[key] = FC['logFC']
        FC_dict[key] = FC['logFC']
        
    FC_df = pd.DataFrame(FC_dict)
        
    return FC_df


def new_corr_stats(corr_values, Prefer, N):
    """
    corr_values: pandas Series with MultiIndex (i,j) and correlation values
    Prefer: pandas Series or dict mapping sample -> group (True/False or labels)
    N: int, sample size used in the sd formula
    Returns: dict with keys including z_w, z_b, z, p_val, log_p, minus_log10_p, and diagnostics
    """
    import numpy as np
    import pandas as pd
    from scipy.stats import norm

    result_dict = {}

    # normalize Prefer to a dict for fast lookup
    try:
        pref_dict = Prefer.to_dict()
    except Exception:
        # if Prefer already a dict-like
        pref_dict = dict(Prefer)

    # Prepare corr_df
    corr_df = pd.DataFrame({'corr': corr_values.values},
                            index=corr_values.index).reset_index()
    # corr_df columns: level_0, level_1, corr
    corr_df.columns = ['level_0', 'level_1', 'corr']

    # determine group membership (True/False) safely
    def pref_get(k):
        # fall back to None if key not found
        return pref_dict.get(k, None)

    corr_df['group'] = corr_df.apply(lambda r: pref_get(r['level_0']) == pref_get(r['level_1']), axis=1)

    # compute means per group if available
    if corr_df['group'].nunique() == 0:
        # no grouping info
        result_dict['corr_Within_group'] = np.nan
        result_dict['corr_Between_group'] = np.nan
        result_dict['corr_diff'] = np.nan
    else:
        corr_group_mean = corr_df.groupby(by='group')['corr'].mean()
        # safe indexing: True might be index True/False, ensure both exist
        within = corr_group_mean.get(True, np.nan)
        between = corr_group_mean.get(False, np.nan)
        result_dict['corr_Within_group'] = within
        result_dict['corr_Between_group'] = between
        result_dict['corr_diff'] = within - between

    # Build within / between edge index lists (tuples)
    edges = list(corr_values.index)
    within_edges = [edge for edge in edges if pref_get(edge[0]) == pref_get(edge[1])]
    between_edges = [edge for edge in edges if pref_get(edge[0]) != pref_get(edge[1])]

    # Convert to arrays of correlation values
    # handle case where no edges in a category
    def get_corr_array(edge_list):
        if len(edge_list) == 0:
            return np.array([], dtype=float)
        else:
            # fetch values from corr_values (MultiIndex)
            return np.array([corr_values[edge] for edge in edge_list], dtype=float)

    corr_within = get_corr_array(within_edges)
    corr_between = get_corr_array(between_edges)

    # Fisher z (arctanh) requires values strictly in (-1,1). clip with small epsilon.
    eps_clip = 1e-15
    corr_within_clipped = np.clip(corr_within, -1 + eps_clip, 1 - eps_clip)
    corr_between_clipped = np.clip(corr_between, -1 + eps_clip, 1 - eps_clip)

    # compute z values elementwise (arctanh)
    # handle empty arrays safely
    z_within_arr = np.arctanh(corr_within_clipped) if corr_within_clipped.size > 0 else np.array([])
    z_between_arr = np.arctanh(corr_between_clipped) if corr_between_clipped.size > 0 else np.array([])

    # means
    z_w = np.nan
    z_b = np.nan
    if z_within_arr.size > 0:
        z_w = float(np.nanmean(z_within_arr))
    if z_between_arr.size > 0:
        z_b = float(np.nanmean(z_between_arr))

    result_dict['z_w'] = z_w
    result_dict['z_b'] = z_b
    result_dict['mu'] = np.nan if (np.isnan(z_w) or np.isnan(z_b)) else (z_w - z_b)

    # compute sd safely
    # original formula: sd = sqrt((1/(N-3)) * (1/n_within + 1/n_between))
    n_within = len(corr_within)
    n_between = len(corr_between)
    sd = np.nan
    if (N - 3) > 0 and n_within > 0 and n_between > 0:
        sd = np.sqrt((1.0 / (N - 3.0)) * ((1.0 / n_within) + (1.0 / n_between)))
    result_dict['sd'] = sd

    # z-statistic
    z_stat = np.nan
    if (not np.isnan(result_dict['mu'])) and (sd is not np.nan) and (sd != 0):
        z_stat = result_dict['mu'] / sd
    result_dict['z'] = z_stat

    # compute p-value in log-space to avoid underflow; use upper-tail (one-sided) like original (1 - cdf)
    if z_stat is None or (isinstance(z_stat, float) and (np.isnan(z_stat))):
        result_dict['log_p'] = np.nan
        result_dict['p_val'] = np.nan
        result_dict['minus_log10_p'] = np.nan
    else:
        # log survival function (logsf) is stable for large z
        log_p = norm.logsf(z_stat)  # log(1 - cdf(z))
        try:
            p_val = float(np.exp(log_p))
        except OverflowError:
            p_val = 0.0

        result_dict['log_p'] = float(log_p)
        result_dict['p_val'] = p_val
        # minus log10 p (useful for plotting)
        ln10 = np.log(10.0)
        result_dict['minus_log10_p'] = float(-log_p / ln10)

    # diagnostics
    result_dict['n_within'] = n_within
    result_dict['n_between'] = n_between
    result_dict['n_total_edges'] = len(edges)

    return result_dict
