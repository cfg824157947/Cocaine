#+begin_src jupyter-python :session /jpy:localhost#8888:cham_test2 :kernel VCC :async yes
  import gseapy as gp
  import numpy as np
  import seaborn as sns
  import matplotlib.pyplot as plt


  for file_name in file_list:
      if 'Human' in file_name and 'Kenyon' in file_name:
          print(file_name)
          file_path = f'{DIOPT_dir}/{file_name}'
          diopt_df = pd.read_csv(file_path)
      else:
          continue
      # Filter genes mapping to Human with DIOPT score > 3j
      human_genes = diopt_df[(diopt_df['Species 2'] == 'Human') & (diopt_df['DIOPT Score'] > 5)]['Human Symbol'].unique().tolist()

      print(f"Total unique human genes with DIOPT score > 5: {len(human_genes)}")

      # Run Enrichment for GO Biological Process
      enr = gp.enrichr(gene_list=human_genes,
                      #gene_sets=['GO_Biological_Process_2025'],
                      gene_sets=['KEGG_2026'],
                      organism='human',
                      outdir=None)

      # Display top 10 results
      print(tabulate(enr.results.head(10), headers='keys', tablefmt='org'))

      # Pre-process results for plotting
      df_plot = enr.results.copy()

      # Calculate -log10(Adjusted P-value) for bubble size
      df_plot['nlog10'] = -np.log10(df_plot['Adjusted P-value'].replace(0, 1e-300))

      # Ensure numeric types
      df_plot['Combined Score'] = pd.to_numeric(df_plot['Combined Score'], errors='coerce')
      df_plot['Odds Ratio'] = pd.to_numeric(df_plot['Odds Ratio'], errors='coerce')

      # Sort and take top 20 terms by Combined Score
      df_plot = df_plot.sort_values('Adjusted P-value', ascending=True).head(20)

      # Generate Bubble Plot using Seaborn
      plt.figure(figsize=(10, 8))
      sns.set_style("whitegrid")

      # Use scatterplot for bubble plot
      scatter = sns.barplot(data=df_plot, 
                                hue='Odds Ratio', 
                                y='Term', 
                                x='nlog10', 
                                palette='viridis',
                                alpha=0.8)

      plt.title(f"GO Biological Process Enrichment {file_name} (DIOPT > 5)", fontsize=15)
      plt.xlabel("-log10(adj.pvalue)", fontsize=12)
      plt.ylabel("GO Term", fontsize=12)

      # Move legend outside
      plt.legend(bbox_to_anchor=(1.05, 1), loc='upper left', borderaxespad=0. , title='Odds Ratio')
      plt.tight_layout()
      plt.show()
