  import pandas as pd
  import re

  # Input file
  SNP_file = "/project/zduren/durenlab/palmetto/cham/cocain/GWAS/GWAS_from_Trudy.tsv"


  # Read GWAS table
  snp_df = pd.read_csv(SNP_file, sep="\t", index_col=0)

  # Reset index so Trait becomes a normal column
  snp_df = snp_df.reset_index()

  # Keep only rows with a valid Variant ID
  snp_df = snp_df[snp_df["Variant ID"].notna()].copy()

  # Parse Variant ID format like: 2L_10154525_T_C
  def parse_variant_id(vid):
      parts = str(vid).split("_")
      if len(parts) < 4:
          return None, None, None, None
      chrom = parts[0]
      pos = int(parts[1])
      ref = parts[2]
      alt = parts[3]
      return chrom, pos, ref, alt

  snp_df[["chrom", "pos", "ref", "alt"]] = snp_df["Variant ID"].apply(
      lambda x: pd.Series(parse_variant_id(x))
  )

  # Drop rows that failed parsing
  snp_df = snp_df[snp_df["chrom"].notna() & snp_df["pos"].notna()].copy()

  # BED is 0-based start, half-open end
  snp_df["start"] = snp_df["pos"] - 1
  snp_df["end"] = snp_df["pos"]

  # Optional BED score:
  # use -log10(p-value), clipped to 1000 for BED compatibility
  snp_df["score"] = (-snp_df["Wald Test P-Value"].astype(float).apply(lambda x: 0 if x <= 0 else -__import__("math").log10(x)))
  snp_df["score"] = snp_df["score"].clip(upper=1000)

  # BED name field
  snp_df["name"] = snp_df["Gene Symbol"].astype(str) + "|" + snp_df["Variant ID"].astype(str)

  # Optional strand: not known for SNPs, so use "."
  #snp_df["strand"] = "."

  # Make a BED6 file
  #bed_df = snp_df[["chrom", "start", "end", "name", "score", "strand"]].copy()
  bed_df = snp_df[["chrom", "start", "end", "Variant ID", "Trait", "name", "score", "Additive Effect (β)", "β Standard Error", "Wald Test P-Value",  "FlyBase ID", "Gene Symbol"]].copy()

  # Sort for convenience
  bed_df = bed_df.sort_values(["chrom", "start", "end"]).reset_index(drop=True)


  # Save BED file
