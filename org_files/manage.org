* Description
This document describes the workflow and code used to convert peak-by-cell fragment count files (BED-like TSV) into:
1. A `.h5ad` file for Python-based workflows (e.g., Scanpy)
2. A sparse matrix folder (10X-style) for Seurat (`Read10X()` in R)

* Input File Format
Each input file should be a tab-separated file with the following 5 columns:

| chrom | start | end | barcode | count |
|-------+-------+-----+---------+-------|

Example:
#+begin_example
2L	10001670	10002091	AAATGCCTCCTCACAC-28	2
2L	10001670	10002091	AACCTCACAAGCTACC-13	1
#+end_example

* Output
- `.h5ad` file: stores the AnnData object
- Sparse matrix folder:
  - =matrix.mtx.gz=
  - =barcodes.tsv.gz=
  - =features.tsv.gz=

* Python Script
#+begin_src python :tangle convert_peak_to_adata.py
#!/usr/bin/env python3

import os
import argparse
import pandas as pd
import scipy.sparse as sp
import scipy.io
import anndata as ad
import gzip
import shutil

def convert_tsv_to_adata_and_sparse(tsv_path, h5ad_path, outdir, feature_type="peak"):
    df = pd.read_csv(tsv_path, sep='\t', header=None,
                     names=['chrom', 'start', 'end', 'barcode', 'count'])
    df['region'] = df['chrom'].astype(str) + ":" + df['start'].astype(str) + "-" + df['end'].astype(str)
    df = df[['region', 'barcode', 'count']]

    row_index = pd.Categorical(df['region'])
    col_index = pd.Categorical(df['barcode'])

    X = sp.coo_matrix(
        (df['count'], (row_index.codes, col_index.codes)),
        shape=(len(row_index.categories), len(col_index.categories))
    ).tocsr()

    adata = ad.AnnData(X.T)
    adata.var_names = row_index.categories
    adata.obs_names = col_index.categories
    adata.write(h5ad_path)

    os.makedirs(outdir, exist_ok=True)
    scipy.io.mmwrite(f"{outdir}/matrix.mtx", adata.X.T)
    pd.Series(adata.obs_names).to_csv(f"{outdir}/barcodes.tsv", index=False, header=False)

    features = pd.DataFrame({
        "feature_id": adata.var_names,
        "feature_name": adata.var_names,
        "feature_type": feature_type
    })
    features.to_csv(f"{outdir}/features.tsv", sep="\t", index=False, header=False)

    for fn in ["matrix.mtx", "barcodes.tsv", "features.tsv"]:
        with open(f"{outdir}/{fn}", 'rb') as f_in:
            with gzip.open(f"{outdir}/{fn}.gz", 'wb') as f_out:
                shutil.copyfileobj(f_in, f_out)
        os.remove(f"{outdir}/{fn}")

    print(f"✅ Done: {h5ad_path} and sparse folder: {outdir}")

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Convert BED-style peak count TSV to h5ad and sparse matrix")
    parser.add_argument('--tsv', required=True, help="Input .tsv file with chrom, start, end, barcode, count")
    parser.add_argument('--h5ad', required=True, help="Output path for .h5ad")
    parser.add_argument('--outdir', required=True, help="Output directory for sparse matrix")
    parser.add_argument('--feature_type', default="peak", help="Feature type for features.tsv")

    args = parser.parse_args()
    convert_tsv_to_adata_and_sparse(args.tsv, args.h5ad, args.outdir, args.feature_type)
#+end_src

#+begin_src python
  import scanpy as sc
  import pandas as pd
  h5_file =  "/project/zduren/durenlab/cham/cocain/filtered_feature_bc_matrix.h5"
  raw_adata = sc.read_10x_h5(h5_file)
  import pandas as pd

  # Suppose your AnnData object is called raw_adata
  # Convert var to DataFrame
  var = raw_adata.var.reset_index().rename(columns={"index": "gene_name"})

  # Keep only relevant columns
  df = var[["gene_name", "gene_ids", "interval", "genome"]].copy()

  # Drop rows with missing interval
  df = df.dropna(subset=["interval"])

  # Split interval into chrom, start, end
  df[["chrom", "range"]] = df["interval"].str.split(":", n=1, expand=True)
  df[["start", "end"]] = df["range"].str.split("-", n=1, expand=True)
  df["start"] = df["start"].astype(int)
  df["end"] = df["end"].astype(int)

  # Define TSS as start for now (since direction unknown in adata.var)
  df["tss"] = df["start"]  # if strand info were available, we'd use + or -

  # Build BED columns
  tss_bed = pd.DataFrame({
      "chrom": df["chrom"],
      "start": df["tss"] - 1,      # BED is 0-based, half-open
      "end": df["tss"],
      "gene_name": df["gene_name"],
      "gene_id": df["gene_ids"],
  })

  # Filter to major chromosomes if desired
  main_chroms = ["2L", "2R", "3L", "3R", "X", "Y", "4"]
  tss_bed = tss_bed[tss_bed["chrom"].isin(main_chroms)]

  # Save as BED
  tss_bed.to_csv("TSS_from_adata_dm6.bed", sep="\t", header=False, index=False)

#+end_src

** Separate cluster fragments to cluster sample

#+begin_src shell
# Set input and output directories
in_dir="/project/zduren/durenlab/cham/cocain/ATAC/cluster_fragment/"
out_dir="/project/zduren/durenlab/palmetto/cham/cocain/Final/Fragment/Fixed_Recall_peaks/sample_cluster_fragment/"

mkdir -p "$out_dir"

# Loop over all cluster files
for f in "$in_dir"/*.bed; do
    cluster=$(basename "$f" .bed)
    echo "Processing cluster $cluster ..."
    awk -F'\t' -v outdir="$out_dir" -v c="$cluster" '{
        n = $4
        split(n, a, "-")
        sample = a[length(a)]
        print > (outdir "/cluster" c "_sample_" sample ".bed")
    }' "$f"
done
